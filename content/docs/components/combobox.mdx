Use `Combobox` when users choose from a list and need to filter it first.

## Usage

```tsx
import {
  Combobox,
  ComboboxEmpty,
  ComboboxInput,
  ComboboxItem,
  ComboboxList,
  ComboboxPopup,
} from "@/components/ui/combobox";

const fruits = ["Apple", "Banana", "Orange"];

<Combobox items={fruits}>
  <ComboboxInput aria-label="Select a fruit" placeholder="Select a fruit..." />
  <ComboboxPopup>
    <ComboboxEmpty>No fruits found.</ComboboxEmpty>
    <ComboboxList>
      {(fruit: string) => (
        <ComboboxItem key={fruit} value={fruit}>
          {fruit}
        </ComboboxItem>
      )}
    </ComboboxList>
  </ComboboxPopup>
</Combobox>;
```

## Composition

Use `ComboboxChips`, `ComboboxValue`, and `ComboboxChip` for multiple selection; keep `ComboboxInput` inside the chips container.

## API Reference

`Combobox` wraps [Base UI Combobox](https://base-ui.com/react/components/combobox). Supported Base UI props pass through.

### ComboboxInput Props

<ApiPropsList>
  <ApiProp name="showTrigger" fullType="boolean" defaultValue="true">
    Shows the popup toggle button.
  </ApiProp>
  <ApiProp name="isClearable" fullType="boolean" defaultValue="false">
    Shows a clear action after selection.
  </ApiProp>
</ApiPropsList>

### ComboboxPopup Props

<ApiPropsList>
  <ApiProp
    name="side"
    fullType='"top" | "bottom"'
    simpleType="string"
    defaultValue='"bottom"'
  >
    Sets the preferred side of the input.
  </ApiProp>
  <ApiProp name="sideOffset" fullType="number" defaultValue="0">
    Sets the distance from the input in pixels.
  </ApiProp>
  <ApiProp
    name="align"
    fullType='"start" | "center" | "end"'
    simpleType="string"
    defaultValue='"center"'
  >
    Sets alignment relative to the input.
  </ApiProp>
  <ApiProp name="alignOffset" fullType="number" defaultValue="0">
    Offsets the popup along the alignment axis.
  </ApiProp>
</ApiPropsList>

### ComboboxChips Props

<ApiPropsList>
  <ApiProp
    name="overflowBehavior"
    fullType='"wrap" | "wrap-when-open" | "cutoff"'
    simpleType="string"
    defaultValue='"wrap-when-open"'
  >
    Controls chip wrapping and single-line +X overflow compression.
  </ApiProp>
  <ApiProp name="maxCount" fullType="number">
    Explicit limit on visible chips before showing the +X overflow badge.
  </ApiProp>
</ApiPropsList>
