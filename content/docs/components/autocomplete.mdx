Use `Autocomplete` when users benefit from suggestions but may enter any text.

## Usage

```tsx
import {
  Autocomplete,
  AutocompleteEmpty,
  AutocompleteInput,
  AutocompleteItem,
  AutocompleteList,
  AutocompletePopup,
} from "@/components/ui/autocomplete";

const tags = ["feature", "fix", "bug", "docs"];

<Autocomplete items={tags}>
  <AutocompleteInput placeholder="Search tags..." />
  <AutocompletePopup>
    <AutocompleteEmpty>No tags found.</AutocompleteEmpty>
    <AutocompleteList>
      {(tag: string) => (
        <AutocompleteItem key={tag} value={tag}>
          {tag}
        </AutocompleteItem>
      )}
    </AutocompleteList>
  </AutocompletePopup>
</Autocomplete>;
```

## Composition

Place the input and popup inside `Autocomplete`; use groups, collections, and separators to organize suggestions.

## API Reference

`Autocomplete` wraps [Base UI Autocomplete](https://base-ui.com/react/components/autocomplete). Supported Base UI props pass through.

### AutocompleteInput Props

<ApiPropsList>
  <ApiProp name="showTrigger" fullType="boolean" defaultValue="false">
    Shows a popup toggle button.
  </ApiProp>
  <ApiProp name="showClear" fullType="boolean" defaultValue="false">
    Shows a clear action after text is entered.
  </ApiProp>
  <ApiProp name="isClearable" fullType="boolean" defaultValue="false">
    Alias for `showClear`.
  </ApiProp>
</ApiPropsList>

### AutocompletePopup Props

<ApiPropsList>
  <ApiProp
    name="side"
    fullType='"top" | "bottom"'
    simpleType="string"
    defaultValue='"bottom"'
  >
    Sets the preferred side of the input.
  </ApiProp>
  <ApiProp name="sideOffset" fullType="number" defaultValue="0">
    Sets the distance from the input in pixels.
  </ApiProp>
  <ApiProp
    name="align"
    fullType='"start" | "center" | "end"'
    simpleType="string"
    defaultValue='"center"'
  >
    Sets alignment relative to the input.
  </ApiProp>
  <ApiProp name="alignOffset" fullType="number" defaultValue="0">
    Offsets the popup along the alignment axis.
  </ApiProp>
</ApiPropsList>

### AutocompleteList Props

<ApiPropsList>
  <ApiProp
    name="scrollShadow"
    fullType='"vertical" | "horizontal" | "both" | "none"'
    simpleType="string"
    defaultValue='"vertical"'
  >
    Shows gradients for hidden scrollable content.
  </ApiProp>
  <ApiProp name="hideScrollbar" fullType="boolean" defaultValue="false">
    Hides the scrollbar while preserving scrolling.
  </ApiProp>
</ApiPropsList>
